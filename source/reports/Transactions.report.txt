   '  TransactionsDialogBox form.
   
   Dim intX As Integer
   Dim qdf As Variant
   
   '  Don't open report if TransactionDialogBox form isn't loaded
   If Not (IsLoaded("TransactionsDialogBox")) Then
      Cancel = True

         & "the Transactions Dialog Box in Form view.", 48, _
         "Must Open Dialog Box"
      Exit Sub
   End If
   
   Set dbsReport = CurrentDb()
