Private Sub lock_records_Click()
On Error GoTo lock_records_Err
    DoCmd.SetWarnings False
    DoCmd.RunSQL "update transactions set accounts_locked = true where [transactions].[date]<= [Lock Record Date (all records will be locked before this date)] ", -1
    DoCmd.SetWarnings True
    MsgBox "The specified transactions records are now locked.", vbInformation, "Lock Transaction Records"
    Me.Requery
    DoCmd.Close

lock_records_Exit:
    Exit Sub

lock_records_Err:
    MsgBox Error$
    Resume lock_records_Exit



End Sub

Private Sub unlock_records_Click()
    DoCmd.SetWarnings False
    DoCmd.RunSQL "update transactions set accounts_locked = false", -1
    DoCmd.SetWarnings True
    MsgBox "All transactions records are now unlocked.", vbInformation, "Unlock Records"
    DoCmd.Close
End Sub
