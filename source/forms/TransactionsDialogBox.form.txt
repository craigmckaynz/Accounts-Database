Private Sub export_to_excel_Click()
    'exports data to spreadsheet for current contract
    Dim strFileName As String
    Dim strFilePath As String
    Dim strExcelFile As String
    Dim strContractName As String
    Dim strAreaName As String
    Dim strQueryName As String
    Dim XL As Object
    Dim cell As Range
    Dim max_col As Integer
    'Dim msg, style, title, response
    
    'allows user to select folder in which to save excel file
    Set XL = CreateObject("Excel.Application")
    SaveSetting appname:="mcl", Section:="Startup", Key:="FilePath", setting:=GetBrowse
    strFilePath = GetSetting(appname:="mcl", Section:="Startup", Key:="FilePath", Default:="Not Set")
    strFileName = "mckay_consultants_transactions"
    strQueryName = "Transactionsquery"
    strExcelFile = "" & strFilePath & "" & strFileName & ""
    XL.Visible = True
    DoCmd.TransferSpreadsheet acExport, acSpreadsheetTypeExcel9, "" & strQueryName & "", "" & strExcelFile & ".xls"
    XL.Workbooks.Open ("" & strExcelFile & ".xls")
    XL.ActiveSheet.Name = "" & strFileName & ""
    
    'format all cells to arial 8 size font
    XL.ActiveSheet.Range("a1").CurrentRegion.Select
    With XL.Selection.Font
        .Size = 8
        .Name = "arial"
    End With

    'Format the first row (column headers).
     max_col = XL.ActiveSheet.UsedRange.Columns.Count
     XL.ActiveSheet.Range(XL.ActiveSheet.Cells(1, 1), XL.ActiveSheet.Cells(1, max_col)).Select
     With XL.Selection
         .Font.FontStyle = "Bold"
         .HorizontalAlignment = xlCenter
         .Interior.ColorIndex = 15
     End With
     XL.Selection.Borders(xlDiagonalDown).LineStyle = xlNone
     XL.Selection.Borders(xlDiagonalUp).LineStyle = xlNone
     With XL.Selection.Borders(xlEdgeLeft)
         .LineStyle = xlContinuous
         .Weight = xlThin
         .ColorIndex = xlAutomatic
     End With
     With XL.Selection.Borders(xlEdgeTop)
         .LineStyle = xlContinuous
         .Weight = xlThin
         .ColorIndex = xlAutomatic
     End With
     With XL.Selection.Borders(xlEdgeBottom)
         .LineStyle = xlContinuous
         .Weight = xlThin
         .ColorIndex = xlAutomatic
     End With
     With XL.Selection.Borders(xlEdgeRight)
         .LineStyle = xlContinuous
         .Weight = xlThin
         .ColorIndex = xlAutomatic
     End With
     With XL.Selection.Borders(xlInsideVertical)
         .LineStyle = xlContinuous
         .Weight = xlThin
         .ColorIndex = xlAutomatic
     End With
     
     'Convert headers to title case
     For Each cell In XL.Selection.Cells
         cell = StrConv(cell, vbProperCase)
     Next
     
    'autofit columns
     XL.ActiveSheet.UsedRange.Select
     XL.Selection.Columns.AutoFit
     XL.Cells.Select
     XL.Selection.RowHeight = 15
     XL.Range("A1").Select

End Sub

Private Sub Preview_Click()
    Dim DocName As String
    
    If EndingDate < BeginningDate Then
       MsgBox "The Ending Date must be later than the Beginning Date."
       DoCmd.CancelEvent
       Exit Sub
    End If
    
    If IsNull([Opening_Balance]) Then Opening_Balance = 0
    
    If group_transactions = False Then
        DocName = "Transactions"
    Else
        DocName = "transactions_grouped"
    End If
        
    DoCmd.OpenReport DocName, 2
End Sub
